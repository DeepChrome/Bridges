# Gate 6 — the crash path, on a device (docs/PLAYBOOK.md Phase 6).
#
# Run this against a build with a DELIBERATE throw in Stats.js. It is not part
# of the ordinary walkthrough set for that reason: it proves the boundary by
# breaking something, and the break has to be put back afterwards.
#
#   .\native\tools\walk.ps1 -Flow native\flows\gate6.txt -Device emulator-5554

tap:Monkeynaut
tap:Your name
type:Jared
tap:^Continue$
wait:2
tap:^Skip$
wait:2
tap:^Start from the beginning$
wait:3

# Into the screen that throws.
tap:Your profile
wait:2
tap:^Statistics$
wait:3

# A screen, not a white page. It says what broke and offers two ways out.
shot:90-boundary
expect:^Something broke$
expect:^Try again$
expect:^Back to the path$

# The second button, because trying again on a deterministic throw will not
# help — it goes somewhere else, reading different data.
tap:^Back to the path$
wait:3
shot:91-after-boundary

# …and the crash is on the phone afterwards, where it can be handed over.
tap:Your profile
wait:2
tap:^Settings$
wait:2
shot:92-settings-problem
expect:problem recorded
